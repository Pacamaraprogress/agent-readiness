# Content Signals (contentsignals.org, IETF draft): a usage policy, not a block.
#   search   = search index, links and short excerpts (not AI summaries)
#   ai-input = use in AI answers at query time (RAG, grounding)
#   ai-train = training or fine-tuning models
# A path in front scopes a line, e.g.
# Content-Signal: /research/ search=yes, ai-input=no, ai-train=no
Content-Signal: search=yes, ai-input=yes, ai-train=no
